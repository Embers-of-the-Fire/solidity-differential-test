pragma solidity >=0.8.0;
contract Sar256Mask {
    int256 public last;
    uint256 public count;
    event Did(int256 r, uint256 n);
    function sar256(int256 x, uint256 n) public returns (int256) {
        int256 r = x >> n;
        last = r;
        count += 1;
        emit Did(r, n);
        return r;
    }
}
